Low-Level Design Document
Version 1.0 — Full Technical Specification



1. What This Document Covers
The LLD translates HLD v3 architectural decisions into developer-ready specifications. Every table, service method, error code, queue job, and cache key is defined here. Developers should be able to write code directly from this document without needing to interpret HLD intentions.




2. Database Schema — Complete DDL
PostgreSQL 15. All entities use UUID v4 primary keys except display_order and sequential join table fields. GIN indexes on all JSONB columns used in queries. Row-level locks on stock reservation and coupon redemption.

2.1 Design Rules Applied (from Security Design v1)


2.2 Core Entities

users
CREATE TABLE users (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  mobile          VARCHAR(15) NOT NULL,
  mobile_hash     VARCHAR(64) GENERATED ALWAYS AS (encode(sha256((mobile || 'fs_salt')::bytea), 'hex')) STORED,
  name            VARCHAR(100),
  email           VARCHAR(255),
  firebase_uid    VARCHAR(128) UNIQUE NOT NULL,
  role            VARCHAR(30) NOT NULL DEFAULT 'customer'
                    CHECK (role IN ('customer','superAdmin','inventoryManager','supportAgent')),
  is_deleted      BOOLEAN NOT NULL DEFAULT false,
  deleted_at      TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE UNIQUE INDEX users_mobile_active_idx ON users(mobile) WHERE is_deleted = false;
CREATE INDEX users_firebase_uid_idx ON users(firebase_uid);
-- NOTE: mobile stored plaintext (required for OTP). Masked in all logs as 07XXXXXXX99.
-- On account deletion: mobile → sha256 hash, name/email → NULL, is_deleted = true.

  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),

  colour          VARCHAR(50),
  size_label      VARCHAR(30),       -- display label e.g. 'M (38-40 in)'

  is_deleted      BOOLEAN NOT NULL DEFAULT false,

  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()

3. Repository Pattern — Interface Contracts
Every database interaction goes through a repository class. Controllers call Service methods. Services call Repository methods. Direct DB access from controllers or route handlers is not permitted.


3.1 Base Repository Interface

// types/repository.ts
export interface IBaseRepository<T, CreateDto, UpdateDto> {
  findById(id: string): Promise<T | null>;
  findAll(filters?: Partial<T>): Promise<T[]>;
  create(dto: CreateDto): Promise<T>;
  update(id: string, dto: UpdateDto): Promise<T>;
  softDelete(id: string): Promise<void>;  // sets is_deleted=true, deleted_at=now()
}
 
// Soft delete is applied in every read via a base query wrapper:
// SELECT * FROM {table} WHERE is_deleted = false AND ...
// This is enforced in the Repository base class, not per-method.
 
// IDOR-safe user-scoped read interface (for orders, payments, returns, addresses):
export interface IUserScopedRepository<T> {
  // user_id ALWAYS comes from JWT claims passed down from middleware
  // NEVER from request body or URL parameters
  findByIdForUser(id: string, requestingUserId: string): Promise<T | null>;
  findAllForUser(requestingUserId: string, filters?: object): Promise<T[]>;
}
 
// 404 (not 403) when IDOR check fails: do not reveal resource existence
// Example: order belongs to User B, User A requests it → return null → 404

3.2 Key Repository Class Interfaces

// repositories/order.repository.ts
export class OrderRepository implements IUserScopedRepository<Order> {
 
  // User-scoped: always includes user_id in WHERE clause
  async findByIdForUser(orderId: string, userId: string): Promise<Order | null> {
    return db.query(
      `SELECT * FROM orders WHERE id = $1 AND user_id = $2 AND is_deleted = false`,
      [orderId, userId]   // userId from JWT claims, never from request body
    );
  }
 
  // Admin-only: no user_id filter — role check happens before calling this
  async findByIdAsAdmin(orderId: string): Promise<Order | null> {
    return db.query(

       WHERE user_id=$1 AND channel=$2 AND notification_type=$3

      [userId, channel, type]
    );

    return latest.opted_out_at === null;  // opted_out_at null = still active
  }

export class StockAdjustmentRepository {
 
  async insertAdjustment(dto: CreateAdjustmentDto): Promise<void> {
    await db.transaction(async (trx) => {
      // 1. Lock variant row for update
      const variant = await trx.query(
        `SELECT qty FROM variants WHERE id = $1 FOR UPDATE`, [dto.variantId]
      );
      // 2. Insert adjustment record
      await trx.query(
        `INSERT INTO stock_adjustments
         (variant_id,before_qty,after_qty,reason,notes,adjusted_by)
         VALUES ($1,$2,$3,$4,$5,$6)`,

      );
      // 3. Update variant qty
      await trx.query(
        `UPDATE variants SET qty=$1, updated_at=now() WHERE id=$2`,
        [dto.afterQty, dto.variantId]
      );
    });
  }

3.3 Inventory — Atomic Stock Reservation
// repositories/inventory.repository.ts (stock reservation on add-to-cart)
 
async reserveStock(variantId: string, quantity: number): Promise<ReservationResult> {
  return db.transaction(async (trx) => {
    // Row-level lock: prevents concurrent over-sell
    const variant = await trx.query(
      `SELECT id, qty FROM variants WHERE id = $1 AND is_deleted = false FOR UPDATE`,
      [variantId]
    );
    if (!variant) throw new AppError(ErrorCode.PRODUCT_NOT_FOUND);
    if (variant.qty < quantity) throw new AppError(ErrorCode.STOCK_UNAVAILABLE);
    
    // Decrement stock atomically
    await trx.query(
      `UPDATE variants SET qty = qty - $1, updated_at = now() WHERE id = $2`,
      [quantity, variantId]
    );
    
    // If qty reaches 0: publish STOCK_DEPLETED event via BullMQ
    const newQty = variant.qty - quantity;
    if (newQty === 0) {
      await stockEventQueue.add('stock_depleted', { variantId });
    }
    return { reserved: true, remainingQty: newQty };
  });
}
 
async releaseReservation(variantId: string, quantity: number): Promise<void> {
  await db.query(
    `UPDATE variants SET qty = qty + $1, updated_at = now() WHERE id = $2`,
    [quantity, variantId]
  );
  // Also publish STOCK_REPLENISHED if qty was 0 before release
}

3.4 Pricing — Server-Side Effective Price
// repositories/catalog.repository.ts
// Pricing precedence: sale_price > price_override > base_price
// ALWAYS calculated server-side. Client-sent price is NEVER used.
 
async getEffectivePrice(variantId: string): Promise<PriceResult> {
  const variant = await db.query(
    `SELECT base_price, price_override, sale_price, sale_active FROM variants WHERE id = $1`,
    [variantId]
  );
  const effectivePrice =
    variant.sale_active && variant.sale_price != null ? variant.sale_price
    : variant.price_override != null                  ? variant.price_override
    : variant.base_price;
 
  return {
    effectivePrice,
    originalPrice: variant.base_price,
    isOnSale: variant.sale_active && variant.sale_price != null,
    discountAmount: variant.base_price - effectivePrice,
  };
}

// total = taxable_amount + gst_amount

// Error response shape (all environments):
// NOTE: no stack trace, no internal service names, no DB error messages
// {
//   success: false,
//   error: {
//     code: 'COUPON_MIN_ORDER',          // structured code — frontend can act on it
//     message: 'Add ₹50 more to use this coupon',  // human-readable only
//     meta: { shortfall: 50, minOrder: 799 }  // safe metadata (no internal details)
//   }
// }
 
export function globalErrorHandler(err: unknown, req: Request, res: Response) {
  if (err instanceof AppError) {
    Sentry.captureException(err.originalError ?? err, {
      extra: { code: err.code, path: req.path },
      // PII scrubbing applied before Sentry capture — see 6.3
    });
    const status = HTTP_STATUS_MAP[err.code] ?? 500;
    return res.status(status).json({
      success: false,
      error: { code: err.code, message: CLIENT_MESSAGES[err.code], meta: err.meta ?? null }
    });
  }
  // Unexpected error: log to Sentry, return generic 500
  Sentry.captureException(err);
  return res.status(500).json({
    success: false,
    error: { code: 'INTERNAL_ERROR', message: 'Something went wrong. Please try again.', meta: null }
  });
}

// sentry.config.ts
Sentry.init({
  dsn: process.env.SENTRY_DSN,
  environment: process.env.NODE_ENV,
  beforeSend(event) {
    // Scrub PII fields from request body, query, headers before sending
    const PII_KEYS = ['mobile','phone','email','address','otp','token',
                      'password','secret','card','cvv','upi','firebase_uid'];
    if (event.request?.data) {
      event.request.data = scrubObject(event.request.data, PII_KEYS);
    }
    if (event.extra) {
      event.extra = scrubObject(event.extra, PII_KEYS);
    }
    return event;
  },
});
 
// Mobile number masking for logs (NOT Sentry — for application logs):
// '9876543210' → '98XXXXXX10'
function maskMobile(mobile: string): string {
  if (mobile.length < 4) return 'XXXXX';
  return mobile.slice(0,2) + 'X'.repeat(mobile.length-4) + mobile.slice(-2);
}
// Rule: mobile is NEVER logged in plain text. Always maskMobile() before any logger call.

// Pattern 1: Explicit DEL on admin save
// Used for: nav, homepage config, collections
// Never passive expiry for config — change must be immediate.
 
async function updateNavItems(items: NavItem[], adminId: string): Promise<void> {

}
 
// Pattern 2: Event-driven invalidation
// Used for: collections after product save
 
async function onProductSaved(productId: string, changedFields: string[]) {
  const affectedCollections = await collectionRepo.findSmartByRuleFields(changedFields);
  for (const collection of affectedCollections) {
    await collectionService.rebuildSmartCollection(collection.id);
    await redis.del(`collection:${collection.slug}`);
  }
}
 
// Pattern 3: Cache stampede prevention
// Used for: collection page load under high traffic
// Problem: cache expires → 100 simultaneous requests hit DB
 
async function getCollectionWithLock(slug: string): Promise<CollectionPage> {
  const cached = await redis.get(`collection:${slug}`);
  if (cached) return JSON.parse(cached);
 
  // Acquire lock: only one request rebuilds cache
  const lockKey = `lock:collection:${slug}`;
  const acquired = await redis.set(lockKey, '1', 'EX', 5, 'NX');
  if (!acquired) {
    // Another request is building cache: wait briefly + retry
    await sleep(200);
    return getCollectionWithLock(slug);
  }
  try {
    const data = await buildCollectionData(slug);
    await redis.setex(`collection:${slug}`, 3600, JSON.stringify(data));
    return data;
  } finally {
    await redis.del(lockKey);
  }
}

9.2 Pagination Convention
// Query parameters:
// ?page=1&limit=24&sort=created_at:desc&filter[fabric]=linen&filter[sale_active]=true
 
// Defaults:
// page: 1, limit: 24 (product listings), 10 (admin lists), 50 (analytics)
// sort: created_at:desc
 
// Cursor-based pagination for high-volume lists (notification_log, event_log):
// GET /notifications?cursor={last_id}&limit=50
// Response: { data: [...], nextCursor: '{last_id}' | null }
 
// Filter convention: filter[field]=value
// Multiple values: filter[status]=PLACED,CONFIRMED (comma-separated)
// Range: filter[created_at_from]=2025-01-01&filter[created_at_to]=2025-01-31

9.3 Versioning Strategy
// URL-based versioning: /api/v1/...
// Current: v1 only. v2 introduced when breaking changes required.
// Breaking change = removing a field, changing a type, requiring a new mandatory field.
// Adding optional fields or new endpoints: NOT a breaking change — no version bump.
 
// Deprecation policy: v1 maintained for 6 months after v2 launch.
// Deprecation header: Sunset: Sat, 01 Jan 2027 00:00:00 GMT

9.4 Auth Header Convention
// Cookie-based auth (primary — web/PWA):
// Cookie: jwt=<token>; HttpOnly; Secure; SameSite=Strict
// CSRF token in header: X-CSRF-Token: <value>
//   (required on all POST/PATCH/DELETE requests)
 
// API clients (Razorpay webhooks etc.):
// No cookie — signature validation instead (HMAC SHA256)
 
// Admin panel: same cookie auth + additional MFA claim in JWT
// MFA claim: { mfa_verified: true, mfa_verified_at: timestamp }
// If mfa_verified_at > 30 min ago: re-prompt MFA
 
// Rate limit headers returned on all responses:
// X-RateLimit-Limit: 500
// X-RateLimit-Remaining: 487
